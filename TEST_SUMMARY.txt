KEY DESIGN PRINCIPLE: is_ready() POLLING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

All memory operations follow this latency-insensitive pattern:

    memory.read_port::<0>().read(address);     // Issue request
    loop {                                      // Poll until ready
        clk.tick().await;
        if memory.read_port::<0>().is_ready() {
            break;
        }
    }
    let data = memory.read_port::<0>().data(); // Get data

This pattern works identically whether READ_LAT is 1, 2, 5, or 10 cycles!
